cmake_minimum_required(VERSION 3.20)

# プロジェクト設定
project(NyaMeshEditorDebug 
    VERSION 1.0.0 
    DESCRIPTION "NyaMesh Editor DEBUG - Console Application"
    LANGUAGES CXX
)

# C++標準設定
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

# プロジェクトルート設定（安全なパス解決）
get_filename_component(PROJECT_ROOT "${CMAKE_CURRENT_SOURCE_DIR}" ABSOLUTE)
while(NOT EXISTS "${PROJECT_ROOT}/cores_v23")
    get_filename_component(PROJECT_ROOT "${PROJECT_ROOT}/.." ABSOLUTE)
    if("${PROJECT_ROOT}" STREQUAL "/")
        message(FATAL_ERROR "Could not find project root (cores_v23 directory not found)")
    endif()
endwhile()
message(STATUS "Project Root: ${PROJECT_ROOT}")

# ===============================================
# 依存関係設定
# ===============================================

# Qt6検索・設定
find_package(Qt6 REQUIRED COMPONENTS Core Widgets Gui)
set(CMAKE_AUTOMOC ON)

# nlohmann::json取得（FetchContent使用）
include(FetchContent)

FetchContent_Declare(
    nlohmann_json
    URL https://github.com/nlohmann/json/releases/download/v3.11.2/json.tar.xz
    URL_HASH SHA256=8c4b26bf4b422252e13f332bc5e388ec0ab5c3443d24399acb675e68278d341f
    DOWNLOAD_EXTRACT_TIMESTAMP true
)

FetchContent_MakeAvailable(nlohmann_json)

# ===============================================
# インクルードディレクトリ設定

# ===============================================
# ターゲット定義
# ===============================================

# 実行ファイル作成（コンソールアプリケーション）
add_executable(NyaMeshEditorDebug
    ${NYAMESH_EDITOR_SOURCES}
    ${CORES_V23_SOURCES}
    ${NYAMESH_SOURCES}
    ${DEBUG_MAIN_SOURCE}
)

# ターゲットプロパティ設定
set_target_properties(NyaMeshEditorDebug PROPERTIES
    OUTPUT_NAME "NyaMeshEditor_v23_Debug"
    CXX_STANDARD 20
    CXX_STANDARD_REQUIRED ON
    WIN32_EXECUTABLE FALSE  # コンソールアプリケーション（重要！）
)

# ===============================================
# リンク設定
# ===============================================

# Qt6ライブラリリンク
target_link_libraries(NyaMeshEditorDebug
    Qt6::Core
    Qt6::Widgets  
    Qt6::Gui
    nlohmann_json::nlohmann_json
)

# Windows固有設定
if(WIN32)
    # デバッグ用コンソール設定
    set_target_properties(NyaMeshEditorDebug PROPERTIES
        WIN32_EXECUTABLE FALSE
        LINK_FLAGS "/SUBSYSTEM:CONSOLE"
    )
endif()

# Linux固有設定  
if(UNIX AND NOT APPLE)
    target_link_libraries(NyaMeshEditorDebug
        pthread
        dl
    )
endif()

# ===============================================
# コンパイル設定
# ===============================================

# デバッグ用コンパイル定義
target_compile_definitions(NyaMeshEditorDebug PRIVATE
    NYAMESH_EDITOR_DEBUG
    NYAMESH_EDITOR_VERSION="1.0.0-DEBUG"
    NYAMESH_6CORE_INTEGRATED
)

# 警告レベル設定（デバッグ用）
if(MSVC)
    target_compile_options(NyaMeshEditorDebug PRIVATE /W3 /utf-8 /Zi)
else()
    target_compile_options(NyaMeshEditorDebug PRIVATE -Wall -Wextra -Wpedantic -g)
endif()
