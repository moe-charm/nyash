# ===============================================
# NyaMesh Editor - CMake Build Configuration
# ===============================================
# 
# NyaMesh統合エディタアプリケーション:
# - QWidget + UltraLightCoreBase 多重継承
# - 6コア統合 Intent駆動アーキテクチャ
# - VSCode風モダンUI
# - P2P Ready 分散編集対応
# 
# ビルド要件:
# - Qt6 (Core, Widgets, Gui)  
# - nlohmann::json (FetchContent)
# - C++17以上
# - Windows/Linux対応
# 
# Generated: 2025-07-29
# ===============================================

cmake_minimum_required(VERSION 3.20)

# プロジェクト設定
project(NyaMeshEditor 
    VERSION 1.0.0 
    DESCRIPTION "NyaMesh Editor - Intent-Driven Integrated Editor"
    LANGUAGES CXX
)

# C++標準設定
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

# プロジェクトルート設定（安全なパス解決）
# ビルドディレクトリから実際のプロジェクトルートを計算
get_filename_component(PROJECT_ROOT "${CMAKE_CURRENT_SOURCE_DIR}" ABSOLUTE)
while(NOT EXISTS "${PROJECT_ROOT}/cores_v23")
    get_filename_component(PROJECT_ROOT "${PROJECT_ROOT}/.." ABSOLUTE)
    if("${PROJECT_ROOT}" STREQUAL "/")
        message(FATAL_ERROR "Could not find project root (cores_v23 directory not found)")
    endif()
endwhile()
message(STATUS "Project Root: ${PROJECT_ROOT}")

# ===============================================
# 依存関係設定
# ===============================================

# Qt6検索・設定
find_package(Qt6 REQUIRED COMPONENTS Core Widgets Gui)
set(CMAKE_AUTOMOC ON)

# nlohmann::json取得（FetchContent使用）
include(FetchContent)

FetchContent_Declare(
    nlohmann_json
    URL https://github.com/nlohmann/json/releases/download/v3.11.2/json.tar.xz
    URL_HASH SHA256=8c4b26bf4b422252e13f332bc5e388ec0ab5c3443d24399acb675e68278d341f
    DOWNLOAD_EXTRACT_TIMESTAMP true
)

FetchContent_MakeAvailable(nlohmann_json)

# ===============================================
# インクルードディレクトリ設定
# ===============================================

include_directories(
    ${PROJECT_ROOT}
    ${PROJECT_ROOT}/nyamesh/core
    ${PROJECT_ROOT}/cores_v23/base
    ${PROJECT_ROOT}/cores_v23/editor
    ${PROJECT_ROOT}/cores_v23/settings
    ${PROJECT_ROOT}/cores_v23/filesystem
    ${PROJECT_ROOT}/cores_v23/ui
    ${PROJECT_ROOT}/cores_v23/localization
    ${PROJECT_ROOT}/cores_v23/quickaccess
    ${PROJECT_ROOT}/cores_v23/app
    ${Qt6Core_INCLUDE_DIRS}
    ${Qt6Widgets_INCLUDE_DIRS}
    ${Qt6Gui_INCLUDE_DIRS}
)

# ===============================================
# ソースファイル定義
# ===============================================

# NyaMesh Editor メインアプリケーション
set(NYAMESH_EDITOR_SOURCES
    ${PROJECT_ROOT}/cores_v23/app/nyamesh_editor.cpp
    ${PROJECT_ROOT}/cores_v23/app/nyamesh_editor.h
)

# 6コア統合ソース
set(CORES_V23_SOURCES
    # Base Classes
    ${PROJECT_ROOT}/cores_v23/base/UltraLightCoreBase_Integrated.cpp
    ${PROJECT_ROOT}/cores_v23/base/UltraLightCoreBase_Integrated.h
    
    # Editor Core
    ${PROJECT_ROOT}/cores_v23/editor/EditorCore_v23_Integrated.cpp
    ${PROJECT_ROOT}/cores_v23/editor/EditorCore_v23_Integrated.h
    
    # Settings Core
    ${PROJECT_ROOT}/cores_v23/settings/SettingsCore_v23_Integrated.cpp
    ${PROJECT_ROOT}/cores_v23/settings/SettingsCore_v23_Integrated.h
    
    # FileSystem Core
    ${PROJECT_ROOT}/cores_v23/filesystem/FileSystemCore_v23_Integrated.cpp
    ${PROJECT_ROOT}/cores_v23/filesystem/FileSystemCore_v23_Integrated.h
    
    # UI Core
    ${PROJECT_ROOT}/cores_v23/ui/UICore_v23_Integrated.cpp
    ${PROJECT_ROOT}/cores_v23/ui/UICore_v23_Integrated.h
    
    # Localization Core
    ${PROJECT_ROOT}/cores_v23/localization/LocalizationCore_v23_Integrated.cpp
    ${PROJECT_ROOT}/cores_v23/localization/LocalizationCore_v23_Integrated.h
    
    # QuickAccess Core  
    ${PROJECT_ROOT}/cores_v23/quickaccess/QuickAccessCore_v23_Integrated.cpp
    ${PROJECT_ROOT}/cores_v23/quickaccess/QuickAccessCore_v23_Integrated.h
)

# NyaMesh Transport Layer
set(NYAMESH_SOURCES
    ${PROJECT_ROOT}/nyamesh/core/nyamesh.cpp
    ${PROJECT_ROOT}/nyamesh/core/nyamesh.h
    ${PROJECT_ROOT}/nyamesh/messages/CoreMessages.cpp
    ${PROJECT_ROOT}/nyamesh/messages/CoreMessages.h
    ${PROJECT_ROOT}/nyamesh/transport/NyaMesh2TransportAdapter.cpp
    ${PROJECT_ROOT}/nyamesh/transport/NyaMesh2TransportAdapter.h
    ${PROJECT_ROOT}/nyamesh/transport/InProcessTransport.cpp
    ${PROJECT_ROOT}/nyamesh/transport/InProcessTransport.h
    ${PROJECT_ROOT}/nyamesh/debug/NyaMeshDebugLogger.cpp
    ${PROJECT_ROOT}/nyamesh/debug/NyaMeshDebugLogger.h
    ${PROJECT_ROOT}/nyamesh/intents/IntentCommonOptions.cpp
    ${PROJECT_ROOT}/nyamesh/intents/IntentCommonOptions.h
)

# テストアプリケーション作成（メイン関数）
set(TEST_MAIN_SOURCE
    ${CMAKE_CURRENT_BINARY_DIR}/test_nyamesh_editor_main.cpp
)

# メイン関数自動生成
file(WRITE ${TEST_MAIN_SOURCE} "
/**
 * @file test_nyamesh_editor_main.cpp
 * @brief NyaMesh Editor テストアプリケーション - 自動生成
 */

#include \"cores_v23/app/nyamesh_editor.h\"
#include <QApplication>
#include <iostream>

int main(int argc, char* argv[]) {
    std::cout << \"=== NyaMesh Editor v23 Starting ===\" << std::endl;
    
    QApplication app(argc, argv);
    app.setApplicationName(\"NyaMesh Editor\");
    app.setApplicationVersion(\"1.0.0\");
    app.setOrganizationName(\"NyaMesh Project\");
    
    try {
        auto editor = std::make_unique<charmcode::NyaMeshEditor>();
        editor->show();
        
        std::cout << \"NyaMesh Editor launched successfully!\" << std::endl;
        std::cout << \"Features: 6-Core Integration, Intent-Driven UI, VSCode-style Interface\" << std::endl;
        
        int result = app.exec();
        
        std::cout << \"=== NyaMesh Editor Shutdown ===\" << std::endl;
        return result;
        
    } catch (const std::exception& e) {
        std::cerr << \"ERROR: \" << e.what() << std::endl;
        return 1;
    }
}
")

# ===============================================
# ターゲット定義
# ===============================================

# 実行ファイル作成
add_executable(NyaMeshEditor
    ${NYAMESH_EDITOR_SOURCES}
    ${CORES_V23_SOURCES}
    ${NYAMESH_SOURCES}
    ${TEST_MAIN_SOURCE}
)

# ターゲットプロパティ設定
set_target_properties(NyaMeshEditor PROPERTIES
    OUTPUT_NAME "NyaMeshEditor_v23"
    CXX_STANDARD 20
    CXX_STANDARD_REQUIRED ON
    WIN32_EXECUTABLE TRUE  # Windows GUI アプリケーション
)

# ===============================================
# リンク設定
# ===============================================

# Qt6ライブラリリンク
target_link_libraries(NyaMeshEditor
    Qt6::Core
    Qt6::Widgets  
    Qt6::Gui
    nlohmann_json::nlohmann_json
)

# Windows固有設定
if(WIN32)
    # Windows GUI アプリケーション設定
    set_target_properties(NyaMeshEditor PROPERTIES
        WIN32_EXECUTABLE TRUE
        LINK_FLAGS "/SUBSYSTEM:WINDOWS"
    )
    
    # Visual Studio フォルダ整理
    source_group("NyaMesh Editor" FILES ${NYAMESH_EDITOR_SOURCES})
    source_group("Cores v23" FILES ${CORES_V23_SOURCES})
    source_group("NyaMesh Transport" FILES ${NYAMESH_SOURCES})
    source_group("Generated" FILES ${TEST_MAIN_SOURCE})
endif()

# Linux固有設定  
if(UNIX AND NOT APPLE)
    # Linux GUI アプリケーション設定
    target_link_libraries(NyaMeshEditor
        pthread
        dl
    )
endif()

# ===============================================
# コンパイル設定
# ===============================================

# コンパイル定義
target_compile_definitions(NyaMeshEditor PRIVATE
    QT_NO_DEBUG_OUTPUT
    NYAMESH_EDITOR_VERSION="1.0.0"
    NYAMESH_6CORE_INTEGRATED
)

# 警告レベル設定
if(MSVC)
    target_compile_options(NyaMeshEditor PRIVATE /W3 /utf-8)
else()
    target_compile_options(NyaMeshEditor PRIVATE -Wall -Wextra -Wpedantic)
endif()

# ===============================================
# インストール・デプロイ設定
# ===============================================

# 実行ファイルインストール
install(TARGETS NyaMeshEditor
    RUNTIME DESTINATION bin
    LIBRARY DESTINATION lib
    ARCHIVE DESTINATION lib
)

# 設定ファイル・リソースのインストール
install(FILES
    ${PROJECT_ROOT}/FEATURES_GUIDE.txt
    ${PROJECT_ROOT}/SHORTCUTS.txt
    DESTINATION docs
    OPTIONAL
)

# ===============================================
# ビルド情報表示
# ===============================================

message(STATUS "=== NyaMesh Editor Build Configuration ===")
message(STATUS "Build Type: ${CMAKE_BUILD_TYPE}")
message(STATUS "C++ Standard: ${CMAKE_CXX_STANDARD}")
message(STATUS "Qt Version: ${Qt6_VERSION}")
message(STATUS "Project Root: ${PROJECT_ROOT}")
message(STATUS "Target: NyaMeshEditor_v23")
message(STATUS "Features: 6-Core Integration, Intent-Driven, VSCode UI")
message(STATUS "===========================================")
