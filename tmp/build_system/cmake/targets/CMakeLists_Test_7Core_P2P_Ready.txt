# CMakeLists_Test_7Core_P2P_Ready.txt
# 7コアP2P統合テスト用 CMake設定ファイル（AppNyaMeshNode活用版）
# EditorCore + SettingsCore + FileSystemCore + UICore + LocalizationCore + QuickAccessCore + AppNyaMeshNode (7コア完全版) P2P Ready統合テスト

cmake_minimum_required(VERSION 3.16)
project(Test_7Core_P2P_Ready)

# プロジェクトルートディレクトリを明示的に設定
# ビルドディレクトリから実行される場合の親ディレクトリ
set(PROJECT_ROOT ${CMAKE_CURRENT_SOURCE_DIR}/..)

# C++20標準を使用
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

# Qt6パッケージの検索
find_package(Qt6 REQUIRED COMPONENTS Core Widgets)

# MSVCでUTF-8サポートを有効化（日本語コメント対応）
if(MSVC)
    add_compile_options(/utf-8)
endif()

# インクルードディレクトリ（プロジェクトルート基準）
include_directories(
    ${PROJECT_ROOT}
    ${PROJECT_ROOT}/nyamesh/core
    ${PROJECT_ROOT}/nyamesh/transport
    ${PROJECT_ROOT}/nyamesh/messages
    ${PROJECT_ROOT}/cores_v23/base
    ${PROJECT_ROOT}/cores_v23/editor
    ${PROJECT_ROOT}/cores_v23/settings
    ${PROJECT_ROOT}/cores_v23/filesystem
    ${PROJECT_ROOT}/cores_v23/ui
    ${PROJECT_ROOT}/cores_v23/localization
    ${PROJECT_ROOT}/cores_v23/quickaccess
    ${PROJECT_ROOT}/cores_v23/app
)

# nlohmann/json（FetchContent使用）
include(FetchContent)
FetchContent_Declare(
    nlohmann_json
    GIT_REPOSITORY https://github.com/nlohmann/json.git
    GIT_TAG v3.11.3
    GIT_SHALLOW TRUE
)

# JSON_BuildTests OFF: テスト無効化（高速化）
set(JSON_BuildTests OFF CACHE INTERNAL "")
FetchContent_MakeAvailable(nlohmann_json)

# ソースファイル（プロジェクトルート基準）
set(NYAMESH_SOURCES
    # NyaMesh Core (complete)
    ${PROJECT_ROOT}/nyamesh/core/nyamesh.cpp
    ${PROJECT_ROOT}/nyamesh/transport/InProcessTransport.cpp
    ${PROJECT_ROOT}/nyamesh/transport/NyaMesh2TransportAdapter.cpp
    ${PROJECT_ROOT}/nyamesh/debug/NyaMeshDebugLogger.cpp
    ${PROJECT_ROOT}/nyamesh/intents/IntentCommonOptions.cpp
    
    # UltraLightCoreBase統合基盤
    ${PROJECT_ROOT}/cores_v23/base/UltraLightCoreBase_Integrated.cpp
    
    # v23統合コア実装（6コア完全版）
    ${PROJECT_ROOT}/cores_v23/editor/EditorCore_v23_Integrated.cpp
    ${PROJECT_ROOT}/cores_v23/settings/SettingsCore_v23_Integrated.cpp
    ${PROJECT_ROOT}/cores_v23/filesystem/FileSystemCore_v23_Integrated.cpp
    ${PROJECT_ROOT}/cores_v23/ui/UICore_v23_Integrated.cpp
    ${PROJECT_ROOT}/cores_v23/localization/LocalizationCore_v23_Integrated.cpp
    ${PROJECT_ROOT}/cores_v23/quickaccess/QuickAccessCore_v23_Integrated_HeaderCompliant.cpp  # ヘッダー準拠版使用
    
    # AppNyaMeshNode（7コア目）- Composition + Bridge Pattern
    ${PROJECT_ROOT}/cores_v23/app/AppNyaMeshNode.cpp
)

# 実行ファイル作成
add_executable(Test_7Core_P2P_Ready
    ${PROJECT_ROOT}/tests/Test_7Core_P2P_Ready.cpp
    ${NYAMESH_SOURCES}
)

# ライブラリリンク
target_link_libraries(Test_7Core_P2P_Ready
    Qt6::Core
    Qt6::Widgets
    nlohmann_json::nlohmann_json
)

# Qt MOCファイル処理（AppNyaMeshNode.cpp内のQ_OBJECT対応）
set_target_properties(Test_7Core_P2P_Ready PROPERTIES
    AUTOMOC ON
    AUTORCC ON
    AUTOUIC ON
)

# プリプロセッサ定義
target_compile_definitions(Test_7Core_P2P_Ready PRIVATE
    QT_NO_DEBUG_OUTPUT
    NYAMESH_QT_INTEGRATION
    NYAMESH_P2P_READY
)

# 安全性確認メッセージ
message(STATUS "SAFE BUILD: Project root set to ${PROJECT_ROOT}")
message(STATUS "SAFE BUILD: CMake executed from ${CMAKE_CURRENT_SOURCE_DIR}")
message(STATUS "SAFE BUILD: Root CMakeLists.txt will NOT be modified")
message(STATUS "SAFE BUILD: 7-Core P2P Ready Integration TEST (AppNyaMeshNode + Composition Pattern)")
message(STATUS "P2P READY: AppNyaMeshNode (QObject + UltraLightCoreBase) + 6 Cores Integration")
message(STATUS "ARCHITECTURE: Gemini Composition + Bridge Pattern Implementation")
