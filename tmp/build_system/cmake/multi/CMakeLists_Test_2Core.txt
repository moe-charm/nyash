# CMakeLists_Test_2Core.txt
# 2コア統合テスト用 CMake設定ファイル
# EditorCore + SettingsCore 統合テスト

cmake_minimum_required(VERSION 3.16)
project(Test_2Core_Editor_Settings)

# C++20標準を使用
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

# Qt6パッケージの検索
find_package(Qt6 REQUIRED COMPONENTS Core Widgets)

# MSVCでUTF-8サポートを有効化（日本語コメント対応）
if(MSVC)
    add_compile_options(/utf-8)
endif()

# インクルードディレクトリ
include_directories(${CMAKE_CURRENT_SOURCE_DIR}/..)
include_directories(${CMAKE_CURRENT_SOURCE_DIR}/../nyamesh/core)
include_directories(${CMAKE_CURRENT_SOURCE_DIR}/../nyamesh/transport)

# nlohmann/json（FetchContent使用）
include(FetchContent)
FetchContent_Declare(
    nlohmann_json
    GIT_REPOSITORY https://github.com/nlohmann/json.git
    GIT_TAG v3.11.3
    GIT_SHALLOW TRUE
)

# JSON_BuildTests OFF: テスト無効化（高速化）
set(JSON_BuildTests OFF CACHE INTERNAL "")
FetchContent_MakeAvailable(nlohmann_json)

# Source Files - NyaMesh Core (with absolute paths)
set(NYAMESH_CORE_SOURCES
    ${CMAKE_CURRENT_SOURCE_DIR}/../nyamesh/core/nyamesh.cpp
    ${CMAKE_CURRENT_SOURCE_DIR}/../nyamesh/transport/InProcessTransport.cpp
    ${CMAKE_CURRENT_SOURCE_DIR}/../nyamesh/transport/NyaMesh2TransportAdapter.cpp
    ${CMAKE_CURRENT_SOURCE_DIR}/../nyamesh/debug/NyaMeshDebugLogger.cpp
    ${CMAKE_CURRENT_SOURCE_DIR}/../nyamesh/intents/IntentCommonOptions.cpp
)

# Source Files - UltraLightCoreBase
set(ULTRALIGHT_CORE_SOURCES
    ${CMAKE_CURRENT_SOURCE_DIR}/../cores_v23/base/UltraLightCoreBase_Integrated.cpp
)

# Source Files - EditorCore
set(EDITOR_CORE_SOURCES
    ${CMAKE_CURRENT_SOURCE_DIR}/../cores_v23/editor/EditorCore_v23_Integrated.cpp
)

# Source Files - SettingsCore
set(SETTINGS_CORE_SOURCES
    ${CMAKE_CURRENT_SOURCE_DIR}/../cores_v23/settings/SettingsCore_v23_Integrated.cpp
)

# Test Application
set(TEST_SOURCES
    ${CMAKE_CURRENT_SOURCE_DIR}/../tests/Test_2Core_Editor_Settings.cpp
)

# Create Test Executable
add_executable(Test_2Core_Editor_Settings
    ${NYAMESH_CORE_SOURCES}
    ${ULTRALIGHT_CORE_SOURCES}
    ${EDITOR_CORE_SOURCES}
    ${SETTINGS_CORE_SOURCES}
    ${TEST_SOURCES}
)

# ライブラリリンク
target_link_libraries(Test_2Core_Editor_Settings
    Qt6::Core
    Qt6::Widgets
    nlohmann_json::nlohmann_json
)

# Qt MOCファイル処理
set_target_properties(Test_2Core_Editor_Settings PROPERTIES
    AUTOMOC ON
    AUTORCC ON
    AUTOUIC ON
)

# Compiler Definitions
target_compile_definitions(Test_2Core_Editor_Settings PRIVATE
    QT_WIDGETS_LIB
    QT_CORE_LIB
)

# Windows specific settings
if(WIN32)
    set_target_properties(Test_2Core_Editor_Settings PROPERTIES
        WIN32_EXECUTABLE TRUE
    )
endif()

# Print build information
message(STATUS "=== Test_2Core_Editor_Settings Build Configuration ===")
message(STATUS "Testing: EditorCore + SettingsCore integration test")
message(STATUS "Features: Intent communication, Settings management, Real-time sync")
message(STATUS "========================================================")
