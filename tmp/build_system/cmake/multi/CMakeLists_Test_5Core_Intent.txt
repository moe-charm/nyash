# CMakeLists_Test_5Core_Intent.txt
# 5コア統合テスト用 CMake設定ファイル（Intent駆動版）
# EditorCore + SettingsCore + FileSystemCore + UICore + LocalizationCore 統合テスト
# 4コアベース + LocalizationCore追加

cmake_minimum_required(VERSION 3.16)
project(Test_5Core_Add_Localization)

# C++20標準を使用
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

# Qt6パッケージの検索
find_package(Qt6 REQUIRED COMPONENTS Core Widgets)

# MSVCでUTF-8サポートを有効化（日本語コメント対応）
if(MSVC)
    add_compile_options(/utf-8)
endif()

# インクルードディレクトリ
include_directories(
    ${CMAKE_CURRENT_SOURCE_DIR}
    ${CMAKE_CURRENT_SOURCE_DIR}/nyamesh/core
    ${CMAKE_CURRENT_SOURCE_DIR}/nyamesh/transport
    ${CMAKE_CURRENT_SOURCE_DIR}/nyamesh/messages
    ${CMAKE_CURRENT_SOURCE_DIR}/cores_v23/base
    ${CMAKE_CURRENT_SOURCE_DIR}/cores_v23/editor
    ${CMAKE_CURRENT_SOURCE_DIR}/cores_v23/settings
    ${CMAKE_CURRENT_SOURCE_DIR}/cores_v23/filesystem
    ${CMAKE_CURRENT_SOURCE_DIR}/cores_v23/ui
    ${CMAKE_CURRENT_SOURCE_DIR}/cores_v23/localization
)

# nlohmann/json（FetchContent使用）
include(FetchContent)
FetchContent_Declare(
    nlohmann_json
    GIT_REPOSITORY https://github.com/nlohmann/json.git
    GIT_TAG v3.11.3
    GIT_SHALLOW TRUE
)

# JSON_BuildTests OFF: テスト無効化（高速化）
set(JSON_BuildTests OFF CACHE INTERNAL "")
FetchContent_MakeAvailable(nlohmann_json)

# ソースファイル
set(NYAMESH_SOURCES
    # NyaMesh Core (complete)
    nyamesh/core/nyamesh.cpp
    nyamesh/transport/InProcessTransport.cpp
    nyamesh/transport/NyaMesh2TransportAdapter.cpp
    nyamesh/debug/NyaMeshDebugLogger.cpp
    nyamesh/intents/IntentCommonOptions.cpp
    
    # UltraLightCoreBase統合基盤
    cores_v23/base/UltraLightCoreBase_Integrated.cpp
    
    # v23統合コア実装（5コア）
    cores_v23/editor/EditorCore_v23_Integrated.cpp
    cores_v23/settings/SettingsCore_v23_Integrated.cpp
    cores_v23/filesystem/FileSystemCore_v23_Integrated.cpp
    cores_v23/ui/UICore_v23_Integrated.cpp
    cores_v23/localization/LocalizationCore_v23_Integrated.cpp
)

# 実行ファイル作成
add_executable(Test_5Core_Add_Localization
    tests/Test_5Core_Add_Localization.cpp
    ${NYAMESH_SOURCES}
)

# ライブラリリンク
target_link_libraries(Test_5Core_Add_Localization
    Qt6::Core
    Qt6::Widgets
    nlohmann_json::nlohmann_json
)

# Qt MOCファイル処理
set_target_properties(Test_5Core_Add_Localization PROPERTIES
    AUTOMOC ON
    AUTORCC ON
    AUTOUIC ON
)

# プリプロセッサ定義
target_compile_definitions(Test_5Core_Add_Localization PRIVATE
    QT_NO_DEBUG_OUTPUT
    NYAMESH_QT_INTEGRATION
)
